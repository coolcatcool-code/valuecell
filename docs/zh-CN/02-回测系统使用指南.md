## 回测系统概述

### 什么是回测？

**回测 (Backtesting)** 是使用历史数据模拟交易策略的过程，目的是：
- 验证策略的有效性
- 评估风险和收益特征
- 优化参数设置
- 在不冒真实资金风险的情况下测试想法

```python
# 文件: examples/backtest_simple_ma_cross.py
"""
简单的移动平均线交叉策略回测
策略逻辑：
- 短期均线上穿长期均线 → 买入
- 短期均线下穿长期均线 → 卖出
"""

import pandas as pd
import yfinance as yf
from datetime import datetime

# 1. 获取历史数据
def get_data(symbol, start, end):
    data = yf.download(symbol, start=start, end=end)
    return data

# 2. 计算技术指标
def calculate_indicators(data, short_window=20, long_window=50):
    df = data.copy()
    df['SMA_short'] = df['Close'].rolling(window=short_window).mean()
    df['SMA_long'] = df['Close'].rolling(window=long_window).mean()
    return df

# 3. 生成交易信号
def generate_signals(df):
    df['signal'] = 0
    df.loc[df['SMA_short'] > df['SMA_long'], 'signal'] = 1  # 买入信号
    df.loc[df['SMA_short'] < df['SMA_long'], 'signal'] = -1  # 卖出信号
    df['position'] = df['signal'].shift()  # 下一天执行
    return df

# 4. 回测执行
def backtest(df, initial_capital=10000, commission=0.001):
    capital = initial_capital
    position = 0  # 当前持仓
    trades = []

    for i in range(len(df)):
        if pd.isna(df['position'].iloc[i]):
            continue

        signal = df['position'].iloc[i]
        price = df['Close'].iloc[i]
        date = df.index[i]

        if signal == 1 and position == 0:  # 买入
            shares = (capital * 0.95) / price  # 保留5%现金
            cost = shares * price * (1 + commission)
            if cost <= capital:
                position = shares
                capital -= cost
                trades.append({
                    'date': date,
                    'action': 'BUY',
                    'price': price,
                    'shares': shares,
                    'capital': capital
                })

        elif signal == -1 and position > 0:  # 卖出
            proceeds = position * price * (1 - commission)
            capital += proceeds
            trades.append({
                'date': date,
                'action': 'SELL',
                'price': price,
                'shares': position,
                'capital': capital
            })
            position = 0

    # 计算最终价值
    final_price = df['Close'].iloc[-1]
    final_value = capital + (position * final_price if position > 0 else 0)

    return {
        'initial_capital': initial_capital,
        'final_value': final_value,
        'total_return': (final_value - initial_capital) / initial_capital,
        'trades': trades
    }

# 5. 运行回测
if __name__ == "__main__":
    # 获取数据
    data = get_data('BTC-USD', '2023-01-01', '2024-01-01')

    # 计算指标
    df = calculate_indicators(data, short_window=20, long_window=50)

    # 生成信号
    df = generate_signals(df)

    # 执行回测
    result = backtest(df, initial_capital=10000)

    # 输出结果
    print(f"初始资金: ${result['initial_capital']:,.2f}")
    print(f"最终价值: ${result['final_value']:,.2f}")
    print(f"总收益率: {result['total_return']*100:.2f}%")
    print(f"交易次数: {len(result['trades'])}")

    # 输出交易明细
    print("\n交易明细:")
    for trade in result['trades'][:5]:  # 显示前5笔交易
        print(f"{trade['date'].date()} - {trade['action']} - "
              f"价格: ${trade['price']:.2f} - "
              f"数量: {trade['shares']:.4f} - "
              f"剩余资金: ${trade['capital']:.2f}")
```

```python
def clean_data(df):
    """数据清洗和预处理"""
    # 1. 删除缺失值
    df = df.dropna()

    # 2. 删除重复行
    df = df[~df.index.duplicated(keep='first')]

    # 3. 确保按时间排序
    df = df.sort_index()

    # 4. 异常值处理（例如：价格为0）
    df = df[df['Close'] > 0]

    # 5. 计算收益率
    df['returns'] = df['Close'].pct_change()

    # 6. 删除极端异常值（例如：日收益超过50%）
    df = df[df['returns'].abs() < 0.5]

    return df
```

### 3. 特征工程

```python
def calculate_features(df):
    """计算技术指标特征"""
    import ta  # 技术分析库

    # 移动平均线
    df['SMA_20'] = df['Close'].rolling(window=20).mean()
    df['SMA_50'] = df['Close'].rolling(window=50).mean()
    df['EMA_12'] = df['Close'].ewm(span=12).mean()
    df['EMA_26'] = df['Close'].ewm(span=26).mean()

    # 布林带
    df['BB_upper'] = df['SMA_20'] + 2 * df['Close'].rolling(20).std()
    df['BB_lower'] = df['SMA_20'] - 2 * df['Close'].rolling(20).std()

    # RSI
    df['RSI'] = ta.momentum.RSIIndicator(df['Close']).rsi()

    # MACD
    macd = ta.trend.MACD(df['Close'])
    df['MACD'] = macd.macd()
    df['MACD_signal'] = macd.macd_signal()
    df['MACD_diff'] = macd.macd_diff()

    # ATR (Average True Range)
    df['ATR'] = ta.volatility.AverageTrueRange(
        df['High'], df['Low'], df['Close']
    ).average_true_range()

    # 成交量指标
    df['Volume_SMA'] = df['Volume'].rolling(window=20).mean()
    df['Volume_ratio'] = df['Volume'] / df['Volume_SMA']

    return df
```

```python
# 文件: python/valuecell/agents/my_strategy/core.py
from typing import AsyncGenerator, Optional, Dict
from valuecell.core.types import BaseAgent, StreamResponse
from valuecell.core.agent import streaming

class MyStrategyAgent(BaseAgent):
    """自定义策略智能体"""

    async def stream(
        self,
        query: str,
        conversation_id: str,
        task_id: str,
        dependencies: Optional[Dict] = None,
    ) -> AsyncGenerator[StreamResponse, None]:
        """
        执行策略分析和信号生成

        Args:
            query: 用户查询，例如 "分析 BTC-USD 的趋势"
            conversation_id: 对话ID
            task_id: 任务ID
            dependencies: 依赖数据（可选）
        """
        try:
            # 1. 解析查询，提取交易对
            symbol = self._extract_symbol(query)

            yield streaming.message_chunk(
                f"正在分析 {symbol} 的市场数据..."
            )

            # 2. 获取历史数据
            data = await self._fetch_market_data(symbol)

            # 3. 计算技术指标
            data = self._calculate_indicators(data)

            # 4. 生成交易信号
            signal = self._generate_signal(data)

            # 5. 返回结果
            yield streaming.message_chunk(
                f"分析完成！当前信号: {signal['action']}\n"
                f"原因: {signal['reason']}\n"
                f"建议仓位: {signal['position_size']:.2f}%"
            )

            yield streaming.done()

        except Exception as e:
            yield streaming.error(f"策略执行失败: {str(e)}")

    def _generate_signal(self, data):
        """生成交易信号"""
        latest = data.iloc[-1]

        # 示例：多重条件判断
        if (latest['SMA_20'] > latest['SMA_50'] and
            latest['RSI'] < 70 and
            latest['MACD'] > latest['MACD_signal']):
            return {
                'action': '买入',
                'reason': '短期均线上穿长期均线，RSI未超买，MACD金叉',
                'position_size': 50  # 建议50%仓位
            }
        elif (latest['SMA_20'] < latest['SMA_50'] or
              latest['RSI'] > 70):
            return {
                'action': '卖出',
                'reason': '短期均线下穿或RSI超买',
                'position_size': 0
            }
        else:
            return {
                'action': '持有',
                'reason': '当前无明确信号',
                'position_size': None
            }
```

### 5. 回测执行

#### 完整回测类

```python
# 文件: examples/backtest_engine.py
import pandas as pd
import numpy as np
from typing import Dict, List, Optional
from dataclasses import dataclass
from datetime import datetime

@dataclass
class BacktestConfig:
    """回测配置"""
    initial_capital: float = 10000
    commission: float = 0.001  # 手续费率
    slippage: float = 0.0005   # 滑点率
    position_size: float = 0.95  # 单次交易使用的资金比例

@dataclass
class Trade:
    """交易记录"""
    date: datetime
    action: str  # BUY / SELL
    price: float
    shares: float
    commission: float
    value: float

class BacktestEngine:
    """事件驱动回测引擎"""

    def __init__(self, config: BacktestConfig):
        self.config = config
        self.cash = config.initial_capital
        self.position = 0
        self.trades: List[Trade] = []
        self.equity_curve = []

    def execute_signal(self, date, price, signal):
        """执行交易信号"""
        if signal == 1 and self.position == 0:  # 买入
            self._execute_buy(date, price)
        elif signal == -1 and self.position > 0:  # 卖出
            self._execute_sell(date, price)

        # 记录每日权益
        equity = self.cash + (self.position * price if self.position > 0 else 0)
        self.equity_curve.append({
            'date': date,
            'cash': self.cash,
            'position_value': self.position * price if self.position > 0 else 0,
            'total_equity': equity
        })

    def _execute_buy(self, date, price):
        """执行买入"""
        # 计算买入价格（考虑滑点）
        buy_price = price * (1 + self.config.slippage)

        # 计算可买入数量
        available_cash = self.cash * self.config.position_size
        shares = available_cash / buy_price

        # 计算手续费
        commission = shares * buy_price * self.config.commission

        # 总成本
        total_cost = shares * buy_price + commission

        if total_cost <= self.cash:
            self.position = shares
            self.cash -= total_cost

            self.trades.append(Trade(
                date=date,
                action='BUY',
                price=buy_price,
                shares=shares,
                commission=commission,
                value=total_cost
            ))

    def _execute_sell(self, date, price):
        """执行卖出"""
        # 计算卖出价格（考虑滑点）
        sell_price = price * (1 - self.config.slippage)

        # 计算收入
        proceeds = self.position * sell_price
        commission = proceeds * self.config.commission
        net_proceeds = proceeds - commission

        self.cash += net_proceeds

        self.trades.append(Trade(
            date=date,
            action='SELL',
            price=sell_price,
            shares=self.position,
            commission=commission,
            value=proceeds
        ))

        self.position = 0

    def run(self, data, signals):
        """运行回测"""
        for i in range(len(data)):
            date = data.index[i]
            price = data['Close'].iloc[i]
            signal = signals.iloc[i] if not pd.isna(signals.iloc[i]) else 0

            self.execute_signal(date, price, signal)

        return self.get_results()

    def get_results(self):
        """获取回测结果"""
        equity_df = pd.DataFrame(self.equity_curve)

        return {
            'initial_capital': self.config.initial_capital,
            'final_equity': equity_df['total_equity'].iloc[-1],
            'trades': self.trades,
            'equity_curve': equity_df
        }
```

## 性能评估指标

### 基础指标

```python
def calculate_performance_metrics(equity_curve, trades, initial_capital):
    """计算性能指标"""

    # 1. 总收益率
    final_equity = equity_curve['total_equity'].iloc[-1]
    total_return = (final_equity - initial_capital) / initial_capital

    # 2. 年化收益率
    days = (equity_curve['date'].iloc[-1] - equity_curve['date'].iloc[0]).days
    years = days / 365.25
    annual_return = (1 + total_return) ** (1 / years) - 1

    # 3. 最大回撤
    rolling_max = equity_curve['total_equity'].cummax()
    drawdown = (equity_curve['total_equity'] - rolling_max) / rolling_max
    max_drawdown = drawdown.min()

    # 4. 夏普比率（假设无风险利率为2%）
    equity_curve['returns'] = equity_curve['total_equity'].pct_change()
    excess_returns = equity_curve['returns'] - 0.02/252  # 日化无风险利率
    sharpe_ratio = np.sqrt(252) * excess_returns.mean() / excess_returns.std()

    # 5. 胜率
    winning_trades = [t for t in trades if t.action == 'SELL' and
                      _calculate_trade_pnl(t, trades) > 0]
    win_rate = len(winning_trades) / (len(trades) / 2) if trades else 0

    # 6. 盈亏比
    avg_win = np.mean([_calculate_trade_pnl(t, trades) for t in winning_trades]) if winning_trades else 0
    losing_trades = [t for t in trades if t.action == 'SELL' and
                     _calculate_trade_pnl(t, trades) < 0]
    avg_loss = abs(np.mean([_calculate_trade_pnl(t, trades) for t in losing_trades])) if losing_trades else 1
    profit_factor = avg_win / avg_loss if avg_loss > 0 else float('inf')

    return {
        'total_return': total_return,
        'annual_return': annual_return,
        'max_drawdown': max_drawdown,
        'sharpe_ratio': sharpe_ratio,
        'win_rate': win_rate,
        'profit_factor': profit_factor,
        'total_trades': len(trades),
        'winning_trades': len(winning_trades),
        'losing_trades': len(losing_trades)
    }

def _calculate_trade_pnl(sell_trade, all_trades):
    """计算单笔交易的盈亏"""
    # 找到对应的买入交易
    sell_idx = all_trades.index(sell_trade)
    buy_trade = all_trades[sell_idx - 1]

    buy_cost = buy_trade.price * buy_trade.shares + buy_trade.commission
    sell_proceeds = sell_trade.price * sell_trade.shares - sell_trade.commission

    return sell_proceeds - buy_cost
```

## 常见问题和陷阱

### 过拟合 (Overfitting)

**问题**：策略在历史数据上表现完美，但实盘失败

**原因**：
- 过度优化参数
- 使用了未来数据（前视偏差）
- 样本量太小

**解决方案**：
```python
# 1. 训练集/测试集分离
train_data = data['2020-01-01':'2023-01-01']
test_data = data['2023-01-01':'2024-01-01']

# 在训练集上优化参数
best_params = optimize_parameters(train_data)

# 在测试集上验证
test_results = backtest(test_data, best_params)

# 2. Walk-forward 分析
# 滚动窗口训练和测试

# 3. 限制参数数量
# 参数越少越好，避免过度拟合
```

### 前视偏差 (Look-ahead Bias)

**问题**：使用了未来数据做历史决策

**示例**：
```python
# ❌ 错误：使用当天收盘价计算指标并在当天交易
df['SMA'] = df['Close'].rolling(20).mean()
df['signal'] = (df['Close'] > df['SMA']).astype(int)

# ✅ 正确：使用前一天的指标做今天的决策
df['SMA'] = df['Close'].rolling(20).mean()
df['signal'] = (df['Close'].shift(1) > df['SMA'].shift(1)).astype(int)
```

### 幸存者偏差 (Survivorship Bias)

**问题**：只使用当前存在的股票，忽略已退市的股票

**影响**：高估策略收益

**解决**：使用包含退市股票的完整数据集

### 交易成本低估

**常见错误**：
```python
# ❌ 只考虑手续费
cost = shares * price * 0.001

# ✅ 应考虑：
# - 手续费
# - 滑点
# - 市场冲击（大单）
# - 税收
cost = shares * price * (
    commission_rate +  # 手续费
    slippage_rate +    # 滑点
    market_impact_rate # 市场冲击
)
```
