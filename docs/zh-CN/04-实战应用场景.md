```python
"""
动量策略：买入近期表现最好的股票
策略逻辑：
1. 每月初计算过去3个月收益率
2. 买入前20%表现最好的股票
3. 持有1个月后rebalance
"""

import pandas as pd
import yfinance as yf
from valuecell.backtest import BacktestEngine, Strategy

class MomentumStrategy(Strategy):
    """动量策略"""

    def __init__(self, lookback=90, top_pct=0.2, rebalance_freq='M'):
        self.lookback = lookback  # 回看期（天）
        self.top_pct = top_pct    # 选择前N%
        self.rebalance_freq = rebalance_freq  # 调仓频率

    def calculate_signals(self, data):
        """计算交易信号"""
        signals = pd.DataFrame(index=data.index)

        # 计算每只股票的动量（过去N天收益率）
        for symbol in data.columns.levels[0]:
            returns = data[symbol]['Close'].pct_change(self.lookback)
            signals[symbol] = returns

        # 每个时间点选择动量最强的股票
        signals = signals.apply(lambda x: x >= x.quantile(1 - self.top_pct), axis=1)

        return signals
```

```python
from valuecell.portfolio import PortfolioManager, Strategy

class PortfolioOptimizer:
    """多策略组合优化器"""

    def __init__(self, strategies, capital_allocation='risk_parity'):
        self.strategies = strategies
        self.capital_allocation = capital_allocation

    async def optimize_allocation(self, lookback_days=252):
        """优化策略权重"""

        # 1. 回测每个策略
        strategy_returns = {}
        for name, strategy in self.strategies.items():
            results = await self.backtest(strategy, lookback_days)
            strategy_returns[name] = results.daily_returns

        # 2. 计算协方差矩阵
        returns_df = pd.DataFrame(strategy_returns)
        cov_matrix = returns_df.cov()

        # 3. 根据不同方法分配资金
        if self.capital_allocation == 'equal_weight':
            weights = {name: 1/len(self.strategies) for name in self.strategies}

        elif self.capital_allocation == 'risk_parity':
            # 风险平价：每个策略贡献相同风险
            weights = self._calculate_risk_parity_weights(cov_matrix)

        elif self.capital_allocation == 'max_sharpe':
            # 最大化夏普比率
            weights = self._maximize_sharpe(returns_df, cov_matrix)

        elif self.capital_allocation == 'min_variance':
            # 最小方差组合
            weights = self._minimize_variance(cov_matrix)

        return weights

    def _calculate_risk_parity_weights(self, cov_matrix):
        """风险平价权重计算"""
```

### 场景 5：实时风控系统

**应用场景**:
在实盘交易中实时监控风险，自动止损和减仓

```python
# 文件: risk_management/real_time_monitor.py
from valuecell.trading import RiskManager
import asyncio

class RealTimeRiskMonitor:
    """实时风险监控系统"""

    def __init__(self, portfolio_manager):
        self.portfolio = portfolio_manager
        self.risk_limits = {
            'max_drawdown': 0.15,      # 最大回撤15%
            'max_position': 0.10,      # 单一持仓不超过10%
            'max_daily_loss': 0.03,    # 单日最大亏损3%
            'max_leverage': 1.5,       # 最大杠杆1.5倍
            'var_95': 0.05            # VaR(95%) 不超过5%
        }

    async def monitor(self):
        """持续监控风险"""
        while True:
            # 1. 计算当前指标
            metrics = await self.calculate_risk_metrics()

            # 2. 检查违规
            violations = self.check_violations(metrics)

            # 3. 执行风控措施
            if violations:
                await self.handle_violations(violations)

            # 4. 记录日志
            self.log_metrics(metrics)

            # 每分钟检查一次
            await asyncio.sleep(60)

    async def calculate_risk_metrics(self):
        """计算风险指标"""
        positions = await self.portfolio.get_positions()
        equity = await self.portfolio.get_equity()
        nav = await self.portfolio.get_nav_history(days=30)

        return {
            'drawdown': self._calculate_drawdown(nav),
            'daily_pnl': (equity - nav[-2]) / nav[-2],
            'position_concentration': self._calculate_concentration(positions, equity),
            'leverage': self._calculate_leverage(positions, equity),
            'var_95': self._calculate_var(nav, confidence=0.95)
        }

    def check_violations(self, metrics):
        """检查风险违规"""
        violations = []

        for metric, value in metrics.items():
            limit_key = f'max_{metric}' if 'max' in metric else metric
            if limit_key in self.risk_limits:
                if abs(value) > self.risk_limits[limit_key]:
                    violations.append({
                        'metric': metric,
                        'value': value,
                        'limit': self.risk_limits[limit_key],
                        'severity': self._calculate_severity(value, self.risk_limits[limit_key])
                    })

        return violations

    async def handle_violations(self, violations):
        """处理风险违规"""
        for v in violations:
            print(f"⚠️ 风险告警: {v['metric']} = {v['value']:.2%} "
                  f"(限额: {v['limit']:.2%})")

            if v['severity'] == 'critical':
                # 严重违规：平掉所有仓位
                print("🚨 严重违规！平掉所有仓位！")
                await self.portfolio.close_all_positions()

                # 发送告警通知
                await self.send_alert(
                    title="严重风险告警",
                    message=f"{v['metric']} 达到 {v['value']:.2%}，"
                            f"已强制平仓所有仓位",
                    urgency="critical"
                )

            elif v['severity'] == 'high':
                # 高风险：减仓50%
                print("⚠️ 高风险！减仓50%")
                await self.portfolio.reduce_positions(factor=0.5)

                await self.send_alert(
                    title="高风险告警",
                    message=f"{v['metric']} 达到 {v['value']:.2%}，"
                            f"已减仓50%",
                    urgency="high"
                )

            elif v['severity'] == 'medium':
                # 中风险：暂停新开仓
                print("⚠️ 中风险！暂停新开仓")
```

```python
from valuecell.research import FactorAnalyzer
import pandas as pd

class AlphaFactorMiner:
    """Alpha 因子挖掘器"""

    def __init__(self, universe, start_date, end_date):
        self.universe = universe  # 股票池
        self.start_date = start_date
        self.end_date = end_date

    async def mine_factors(self):
        """挖掘有效因子"""

        # 1. 定义候选因子
        factor_definitions = self._define_factors()

        # 2. 批量计算因子值
        print(f"正在计算 {len(factor_definitions)} 个因子...")
        factor_values = await self._calculate_all_factors(factor_definitions)

        # 3. 因子评估
        print("正在评估因子有效性...")
        factor_performance = []

        for factor_name, values in factor_values.items():
            # IC (Information Coefficient)
            ic = await self._calculate_ic(values)

            # ICIR (IC / IC_std)
            icir = ic.mean() / ic.std()

            # 多空收益
            long_short_returns = await self._calculate_long_short_returns(values)

            # 分层测试
            quantile_returns = await self._quintile_analysis(values)

            factor_performance.append({
                'factor': factor_name,
                'ic_mean': ic.mean(),
                'ic_std': ic.std(),
                'icir': icir,
                'long_short_return': long_short_returns.mean(),
                'long_short_sharpe': long_short_returns.mean() / long_short_returns.std() * np.sqrt(252),
                'top_quintile_return': quantile_returns[0],
                'bottom_quintile_return': quantile_returns[4]
            })

        # 4. 排序和筛选
        df = pd.DataFrame(factor_performance)
        df = df.sort_values('icir', ascending=False)

        # 筛选有效因子
        # 标准: ICIR > 1.0, IC均值 > 0.03, 多空夏普 > 1.5
        effective_factors = df[
            (df['icir'] > 1.0) &
            (df['ic_mean'] > 0.03) &
            (df['long_short_sharpe'] > 1.5)
        ]

        print(f"\n发现 {len(effective_factors)} 个有效因子:")
        print(effective_factors.to_string())

        return effective_factors

    def _define_factors(self):

        return {

            'PE': lambda df: 1 / df['PE'],
            'PB': lambda df: 1 / df['PB'],
            'PS': lambda df: 1 / df['PS'],
            'EV_EBITDA': lambda df: df['EBITDA'] / df['EV'],

            # 成长因子
            'revenue_growth': lambda df: df['revenue'].pct_change(4),
            'earnings_growth': lambda df: df['earnings'].pct_change(4),
            'roe_trend': lambda df: df['ROE'] - df['ROE'].shift(4),

            # 动量因子
            'momentum_1m': lambda df: df['Close'].pct_change(20),
            'momentum_3m': lambda df: df['Close'].pct_change(60),
            'momentum_6m': lambda df: df['Close'].pct_change(120),
            'momentum_12m': lambda df: df['Close'].pct_change(250),

            # 反转因子
            'reversal_1m': lambda df: -df['Close'].pct_change(20),

            # 质量因子
            'roe': lambda df: df['ROE'],
            'roa': lambda df: df['ROA'],
            'gross_margin': lambda df: df['gross_margin'],
            'asset_turnover': lambda df: df['revenue'] / df['total_assets'],

            # 波动率因子
            'volatility_20d': lambda df: df['Close'].pct_change().rolling(20).std(),
            'volatility_60d': lambda df: df['Close'].pct_change().rolling(60).std(),

            # 流动性因子
            'turnover': lambda df: df['volume'] / df['shares_outstanding'],
            'amihud_illiquidity': lambda df: abs(df['returns']) / df['volume'],

            # 技术因子
            'rsi': lambda df: self._calculate_rsi(df['Close']),
            'macd': lambda df: self._calculate_macd(df['Close']),

            # 分析师因子
            'analyst_rating': lambda df: df['analyst_rating'],
            'target_price_upside': lambda df: (df['target_price'] - df['Close']) / df['Close'],

            # 情绪因子
            'news_sentiment': lambda df: df['news_sentiment_score'],
            'social_sentiment': lambda df: df['social_media_sentiment'],

            # 组合因子
            'value_momentum': lambda df: (1/df['PE']) * df['Close'].pct_change(60),
            'quality_value': lambda df: df['ROE'] * (1/df['PB']),
        }
```

```python
from valuecell.core.types import BaseAgent, StreamResponse
from valuecell.core.agent import streaming

class OptionsStrategyAgent(BaseAgent):
    """期权策略智能体"""

    async def stream(self, query, conversation_id, task_id, dependencies=None):

            # 解析查询
            symbol = self._extract_symbol(query)

            yield streaming.message_chunk(

            )
```
