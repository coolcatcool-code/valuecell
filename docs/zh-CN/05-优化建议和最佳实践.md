## 系统性能优化

### 1. 异步 I/O 优化

**问题**: Python 同步 I/O 会阻塞事件循环

**解决方案**：

```python
# ❌ 错误：使用同步库
import requests

def get_market_data(symbol):
    response = requests.get(f"https://api.example.com/data/{symbol}")
    return response.json()

# ✅ 正确：使用异步库
import aiohttp

async def get_market_data(symbol):
    async with aiohttp.ClientSession() as session:
        async with session.get(f"https://api.example.com/data/{symbol}") as response:
            return await response.json()

# ✅ 更好：批量并发请求
async def get_multiple_symbols(symbols):
    async with aiohttp.ClientSession() as session:
        tasks = [get_market_data(session, symbol) for symbol in symbols]
        results = await asyncio.gather(*tasks)
        return dict(zip(symbols, results))
```

**性能提升**：
- 单线程处理多个请求
- 100个股票数据：从30秒 → 3秒
- CPU 利用率提升：20% → 80%

```python
import redis

from typing import Optional

class CacheManager:
    """多层缓存管理"""

    def __init__(self):
        self.redis_client = redis.Redis(host='localhost', port=6379, db=0)

    # 1. 内存缓存（最快，容量小）
    @lru_cache(maxsize=1000)
    def get_static_data(self, key):
        """静态数据缓存（不会变化的数据）"""
        return self._fetch_from_db(key)

    # 2. Redis 缓存（快，容量大）
    async def get_market_data(self, symbol, ttl=300):
        """市场数据缓存（5分钟过期）"""
        cache_key = f"market:{symbol}"

        # 先从 Redis 获取
        cached = self.redis_client.get(cache_key)
        if cached:
            return json.loads(cached)

        # 未命中，从 API 获取
        data = await self._fetch_from_api(symbol)

        # 存入 Redis
        self.redis_client.setex(
            cache_key,
            ttl,
            json.dumps(data)
        )

        return data

    # 3. 写穿缓存（Write-through）
    async def update_user_settings(self, user_id, settings):
        """更新用户设置（同时更新缓存和数据库）"""
        # 1. 更新数据库
        await self.db.update_user_settings(user_id, settings)

        # 2. 更新缓存
        cache_key = f"user_settings:{user_id}"
        self.redis_client.setex(
            cache_key,
            3600,  # 1小时
            json.dumps(settings)
        )

    # 4. 缓存预热
    async def warmup_cache(self):
```

```python
class BatchProcessor:
    """批处理器"""

    def __init__(self, batch_size=100, max_wait=1.0):
        self.batch_size = batch_size
        self.max_wait = max_wait
        self.queue = []
        self.last_flush = time.time()

    async def add(self, item):
        """添加item到批次"""
        self.queue.append(item)

        # 达到批次大小或超时，立即处理
        if len(self.queue) >= self.batch_size or \
           time.time() - self.last_flush > self.max_wait:
            await self.flush()

    async def flush(self):
        """处理当前批次"""
        if not self.queue:
            return

        # 批量处理
        await self._process_batch(self.queue)

        # 清空队列
        self.queue = []
        self.last_flush = time.time()

    async def _process_batch(self, items):
        """批量处理逻辑"""
        # 批量插入数据库
        await db.bulk_insert(items)
```

```python
class LLMRouter:
    """智能模型路由器"""

    MODELS = {
        'gpt-4': {
            'cost_per_1k_input': 0.03,
            'cost_per_1k_output': 0.06,
            'capability': 'high'
        },
        'gpt-3.5-turbo': {
            'cost_per_1k_input': 0.001,
            'cost_per_1k_output': 0.002,
            'capability': 'medium'
        },
        'claude-haiku': {
            'cost_per_1k_input': 0.00025,
            'cost_per_1k_output': 0.00125,
            'capability': 'low'
        }
    }

    def route_query(self, query, complexity=None):
        """根据查询复杂度选择模型"""

        if complexity is None:
            complexity = self._estimate_complexity(query)

        if complexity == 'high':
            # 复杂分析：用最强模型
            return 'gpt-4'
        elif complexity == 'medium':
            # 中等任务：平衡性能和成本
            return 'gpt-3.5-turbo'
        else:
            # 简单任务：最便宜模型
            return 'claude-haiku'

    def _estimate_complexity(self, query):
        """估算查询复杂度"""
        keywords = {
            'high': ['深度分析', '财报', '估值模型', '战略'],
            'medium': ['分析', '对比', '总结'],
            'low': ['查询', '获取', '列出']
        }

        for level, words in keywords.items():
            if any(word in query for word in words):
                return level

        return 'low'

# 使用示例
```

```python
class LLMCache:
    """LLM 响应缓存"""

    def __init__(self):
        self.redis = redis.Redis()

    async def get_or_generate(self, prompt, model='gpt-4'):
        """先查缓存，未命中再调用 LLM"""

        # 生成缓存键（对提示词哈希）
        cache_key = f"llm:{model}:{hashlib.md5(prompt.encode()).hexdigest()}"

        # 查询缓存
        cached = self.redis.get(cache_key)
        if cached:
            print("缓存命中！")
            return json.loads(cached)

        # 未命中，调用 LLM
        print("调用 LLM...")
        response = await self.call_llm(prompt, model)

        # 存入缓存（24小时）
        self.redis.setex(
            cache_key,
            86400,
            json.dumps(response)
        )

        return response
```

```python
    """智能数据管理器"""

    def __init__(self):
        self.cache = {}
        self.sources = {
            'yfinance': {'cost': 0, 'reliability': 0.8, 'latency': 1.0},
            'alpha_vantage': {'cost': 0.01, 'reliability': 0.95, 'latency': 0.5},

        }

    async def get_data(self, symbol, required_reliability=0.9):
        """根据需求选择数据源"""

        # 1. 先查缓存
        if symbol in self.cache:
            return self.cache[symbol]

        # 2. 选择满足可靠性要求的最便宜数据源
        suitable_sources = [
            (name, meta) for name, meta in self.sources.items()
            if meta['reliability'] >= required_reliability
        ]

        # 按成本排序
        suitable_sources.sort(key=lambda x: x[1]['cost'])

            try:
                data = await self._fetch_from_source(source_name, symbol)
                self.cache[symbol] = data
                return data
            except Exception as e:
                print(f"{source_name} 失败：{e}，尝试下一个源...")

        raise Exception("所有数据源都失败了")
```

### 2. 组合权重优化

```python
import cvxpy as cp

def optimize_portfolio_weights(expected_returns, cov_matrix, risk_aversion=1.0):
    """最大化效用函数：E[R] - λ * Var[R]"""

    n = len(expected_returns)
    weights = cp.Variable(n)

    # 预期收益
    portfolio_return = expected_returns @ weights

    # 投资组合方差
    portfolio_risk = cp.quad_form(weights, cov_matrix)

    # 效用函数
    utility = portfolio_return - risk_aversion * portfolio_risk

    # 约束条件
    constraints = [
        cp.sum(weights) == 1,  # 权重和为1
        weights >= 0,          # 不做空
        weights <= 0.2         # 单一持仓不超过20%
    ]

    # 求解
    problem = cp.Problem(cp.Maximize(utility), constraints)
    problem.solve()

    return weights.value
```

### 3. 风险预算

```python
def risk_budgeting_allocation(cov_matrix, risk_budgets):
    """风险预算配置

    Args:
        cov_matrix: 协方差矩阵
        risk_budgets: 每个资产的风险预算（例如：[0.3, 0.3, 0.4]）

    Returns:
        权重向量
    """
    n = len(risk_budgets)

    def objective(weights):
        # 计算边际风险贡献
        portfolio_var = weights @ cov_matrix @ weights
        marginal_contrib = (cov_matrix @ weights) / np.sqrt(portfolio_var)

        # 风险贡献
        risk_contrib = weights * marginal_contrib

        # 误差：实际风险贡献 vs 目标风险预算
        error = np.sum((risk_contrib - risk_budgets)**2)
        return error

    # 约束：权重和为1
    constraints = ({'type': 'eq', 'fun': lambda w: np.sum(w) - 1})

    # 边界：0-1
    bounds = [(0, 1) for _ in range(n)]

    # 初始猜测：等权
    x0 = np.array([1/n] * n)

    # 优化
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)

    return result.x
```

```python
    price: float = Field(None, gt=0)

    @validator('symbol')
    def validate_symbol(cls, v):
        # 额外验证：检查是否在允许列表中
        allowed_symbols = ['AAPL', 'MSFT', 'GOOGL', ...]
        if v not in allowed_symbols:
            raise ValueError(f"Symbol {v} not allowed")
        return v

# 使用
try:
    trade = TradeRequest(
        symbol="AAPL",
        action="buy",
        quantity=100
    )
except ValueError as e:
    print(f"无效输入：{e}")
```

```python
from opentelemetry import trace

trace.get_tracer_provider().add_span_processor(span_processor)

# 使用
async def process_user_input(user_input):
    with tracer.start_as_current_span("process_user_input") as span:
        span.set_attribute("user_id", user_input.user_id)
        span.set_attribute("query", user_input.query)

        # 子 span
        with tracer.start_as_current_span("super_agent"):
            result = await super_agent.run(user_input)

        with tracer.start_as_current_span("planner"):
            plan = await planner.create_plan(result)

        return plan
```
